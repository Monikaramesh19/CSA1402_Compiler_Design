#include <stdio.h>
#include <string.h>
#include <ctype.h>
#define MAX_ID_LEN 31
// Check whether a character is an operator
int isOperator(char ch)
{
    return (ch == '+' || ch == '-' || ch == '*' || ch == '/' ||
            ch == '%' || ch == '=' || ch == '<' || ch == '>' ||
            ch == '!' || ch == '&' || ch == '|');
}

// Check whether a word is a C keyword
int isKeyword(char *word)
{
    char *keywords[] = {
        "int", "float", "char", "double", "void",
        "if", "else", "for", "while", "do",
        "switch", "case", "break", "continue",
        "return", "struct", "const", "long",
        "short", "unsigned", "signed", "static"
    };

    int n = sizeof(keywords) / sizeof(keywords[0]);

    for (int i = 0; i < n; i++)
    {
        if (strcmp(word, keywords[i]) == 0)
            return 1;
    }

    return 0;
}

int main()
{
    FILE *fp;
    char filename[100];

    printf("Enter the input C file name: ");
    scanf("%s", filename);

    fp = fopen(filename, "r");

    if (fp == NULL)
    {
        printf("Error: Cannot open file.\n");
        return 1;
    }

    char ch, next;
    char buffer[100];
    int i;

    printf("\n----- LEXICAL ANALYZER OUTPUT -----\n\n");

    while ((ch = fgetc(fp)) != EOF)
    {
        // Ignore spaces, tabs and new lines
        if (isspace(ch))
            continue;

        // Handle comments
        if (ch == '/')
        {
            next = fgetc(fp);

            // Single-line comment
            if (next == '/')
            {
                while ((ch = fgetc(fp)) != '\n' && ch != EOF)
                    ;
                continue;
            }

            // Multi-line comment
            else if (next == '*')
            {
                char prev = 0;

                while ((ch = fgetc(fp)) != EOF)
                {
                    if (prev == '*' && ch == '/')
                        break;

                    prev = ch;
                }

                continue;
            }

            // Division operator
            else
            {
                ungetc(next, fp);
                printf("Operator    : /\n");
                continue;
            }
        }

        // Identifier or keyword
        if (isalpha(ch) || ch == '_')
        {
            i = 0;
            buffer[i++] = ch;

            while ((ch = fgetc(fp)) != EOF &&
                   (isalnum(ch) || ch == '_'))
            {
                if (i < MAX_ID_LEN)
                    buffer[i++] = ch;
            }

            buffer[i] = '\0';

            if (ch != EOF)
                ungetc(ch, fp);

            if (isKeyword(buffer))
                printf("Keyword     : %s\n", buffer);
            else
                printf("Identifier  : %s\n", buffer);

            continue;
        }

        // Integer or floating-point constant
        if (isdigit(ch))
        {
            i = 0;
            buffer[i++] = ch;

            while ((ch = fgetc(fp)) != EOF &&
                   (isdigit(ch) || ch == '.'))
            {
                buffer[i++] = ch;
            }

            buffer[i] = '\0';

            if (ch != EOF)
                ungetc(ch, fp);

            printf("Constant    : %s\n", buffer);
            continue;
        }

        // String constant
        if (ch == '"')
        {
            i = 0;
            buffer[i++] = ch;

            while ((ch = fgetc(fp)) != EOF && ch != '"')
            {
                if (i < 98)
                    buffer[i++] = ch;
            }

            buffer[i++] = '"';
            buffer[i] = '\0';

            printf("String      : %s\n", buffer);
            continue;
        }

        // Character constant
        if (ch == '\'')
        {
            i = 0;
            buffer[i++] = ch;

            while ((ch = fgetc(fp)) != EOF && ch != '\'')
            {
                if (i < 98)
                    buffer[i++] = ch;
            }

            buffer[i++] = '\'';
            buffer[i] = '\0';

            printf("Character   : %s\n", buffer);
            continue;
        }

        // Two-character operators
        if (isOperator(ch))
        {
            next = fgetc(fp);

            if ((ch == '=' && next == '=') ||
                (ch == '!' && next == '=') ||
                (ch == '<' && next == '=') ||
                (ch == '>' && next == '=') ||
                (ch == '+' && next == '+') ||
                (ch == '-' && next == '-') ||
                (ch == '&' && next == '&') ||
                (ch == '|' && next == '|'))
            {
                printf("Operator    : %c%c\n", ch, next);
            }
            else
            {
                printf("Operator    : %c\n", ch);

                if (next != EOF)
                    ungetc(next, fp);
            }

            continue;
        }

        // Separators and special symbols
        if (ch == '(' || ch == ')' ||
            ch == '{' || ch == '}' ||
            ch == '[' || ch == ']' ||
            ch == ';' || ch == ',' ||
            ch == ':')
        {
            printf("Separator   : %c\n", ch);
            continue;
        }

        // Unknown character
        printf("Unknown     : %c\n", ch);
    }

    fclose(fp);

    printf("\n----- END OF ANALYSIS -----\n");

    return 0;
}
