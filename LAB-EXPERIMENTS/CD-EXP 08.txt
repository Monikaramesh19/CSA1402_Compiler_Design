#include <stdio.h>

int main()
{
    printf("Grammar:\n");
    printf("S -> AaAb | BbBa\n");
    printf("A -> epsilon\n");
    printf("B -> epsilon\n\n");

    printf("----- FOLLOW SETS -----\n");

    // S is the start symbol
    printf("FOLLOW(S) = { $ }\n");

    // A is followed by 'a' in AaAb
    // A is followed by 'b' in BbBa
    printf("FOLLOW(A) = { a, b }\n");

    // B is followed by 'b' in BbBa
    // B is followed by 'a' in AaAb
    printf("FOLLOW(B) = { a, b }\n");

    return 0;
}
