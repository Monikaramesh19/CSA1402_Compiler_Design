#include <stdio.h>
#include <string.h>
#include <ctype.h>

char expression[100];
char temp[10];

int tempCount = 1;

int precedence(char op)
{
    if (op == '*' || op == '/')
        return 2;

    if (op == '+' || op == '-')
        return 1;

    return 0;
}

void generateTAC(char expr[])
{
    char operands[50][10];
    char operators[50];

    int topOperand = -1;
    int topOperator = -1;

    char token[10];
    int i = 0;

    while (expr[i] != '\0')
    {
        if (isalnum(expr[i]))
        {
            int j = 0;

            while (isalnum(expr[i]))
            {
                token[j++] = expr[i++];
            }

            token[j] = '\0';

            strcpy(operands[++topOperand], token);
        }
        else if (expr[i] == '(')
        {
            operators[++topOperator] = expr[i];
            i++;
        }
        else if (expr[i] == ')')
        {
            while (topOperator >= 0 &&
                   operators[topOperator] != '(')
            {
                char op = operators[topOperator--];

                char op2[10], op1[10];

                strcpy(op2, operands[topOperand--]);
                strcpy(op1, operands[topOperand--]);

                sprintf(temp, "t%d", tempCount++);

                printf("%s = %s %c %s\n",
                       temp, op1, op, op2);

                strcpy(operands[++topOperand], temp);
            }

            topOperator--;
            i++;
        }
        else if (expr[i] == '+' ||
                 expr[i] == '-' ||
                 expr[i] == '*' ||
                 expr[i] == '/')
        {
            while (topOperator >= 0 &&
                   precedence(operators[topOperator]) >=
                   precedence(expr[i]))
            {
                char op = operators[topOperator--];

                char op2[10], op1[10];

                strcpy(op2, operands[topOperand--]);
                strcpy(op1, operands[topOperand--]);

                sprintf(temp, "t%d", tempCount++);

                printf("%s = %s %c %s\n",
                       temp, op1, op, op2);

                strcpy(operands[++topOperand], temp);
            }

            operators[++topOperator] = expr[i];
            i++;
        }
        else
        {
            i++;
        }
    }

    while (topOperator >= 0)
    {
        char op = operators[topOperator--];

        char op2[10], op1[10];

        strcpy(op2, operands[topOperand--]);
        strcpy(op1, operands[topOperand--]);

        sprintf(temp, "t%d", tempCount++);

        printf("%s = %s %c %s\n",
               temp, op1, op, op2);

        strcpy(operands[++topOperand], temp);
    }
}

int main()
{
    char lhs;
    char rhs[100];

    printf("Enter assignment statement.\n");
    printf("Example: a=b+c*d\n\n");

    printf("Enter expression: ");
    scanf("%s", expression);

    lhs = expression[0];

    // Find '='
    char *equal = strchr(expression, '=');

    if (equal == NULL)
    {
        printf("Invalid expression.\n");
        return 0;
    }

    strcpy(rhs, equal + 1);

    printf("\n----- THREE ADDRESS CODE -----\n");

    generateTAC(rhs);

    sprintf(temp, "t%d", tempCount - 1);

    printf("%c = %s\n", lhs, temp);

    return 0;
}
